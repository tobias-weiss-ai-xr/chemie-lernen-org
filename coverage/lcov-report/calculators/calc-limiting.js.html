function calcLimitingValue(m1, M1, m2, M2) {
  const n1 = m1 / M1;
  const n2 = m2 / M2;
  const limiting = n1 < n2 ? 1 : 2;
  const name = limiting === 1 ? 'Reagenz 1' : 'Reagenz 2';
  const excess = limiting === 1 ? n2 - n1 : n1 - n2;
  return { n1, n2, limiting, name, excess };
}
